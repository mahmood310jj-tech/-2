Future<T> withRetry<T>(Future<T> Function() fn) async {
  for (int i = 0; i < 3; i++) {
    try {
      return await fn();
    } catch (e) {
      if (i == 2) rethrow;
      await Future.delayed(const Duration(seconds: 2));
    }
  }
  throw Exception('فشل الاتصال');
}

class _LoginPageState extends State<LoginPage> {
  final emailCtrl = TextEditingController();
  final passCtrl = TextEditingController();
  String msg = '';
  bool loading = false;
  Future<void> login() async {
    setState(() {
      loading = true;
      msg = '';
    });
    try {
      await withRetry(() => supabase.auth.signInWithPassword(email: emailCtrl.text.trim(), password: passCtrl.text.trim()));
      if (!mounted) return;
      Navigator.pushReplacement(context, MaterialPageRoute(builder: (_) => const HomePage()));
    } catch (e) {
      if (mounted) setState(() => msg = 'فشل الدخول: $e');
    }
    if (mounted) setState(() => loading = false);
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: bgColor,
      body: SingleChildScrollView(
        padding: const EdgeInsets.all(24),
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            const SizedBox(height: 60),
            const Text('مرحباً بكم في مدرسة المصطفى', textAlign: TextAlign.center, style: TextStyle(fontSize: 24, fontWeight: FontWeight.bold, color: darkGreen)),
            const Text('مدرستنا .. بيت العلم والإبداع', style: TextStyle(color: Colors.grey, fontSize: 14)),
            const SizedBox(height: 28),
            TextField(controller: emailCtrl, decoration: const InputDecoration(labelText: 'البريد الإلكتروني', prefixIcon: Icon(Icons.email, color: mainGreen), border: OutlineInputBorder(borderRadius: BorderRadius.all(Radius.circular(16))), filled: true, fillColor: Colors.white)),

  Future<void> loadAll() async {
    if (!widget.isGuest) {
      try {
        final p = await withRetry(() => supabase.from('profiles').select('full_name, role').eq('id', supabase.auth.currentUser!.id).single());
        if (mounted) setState(() {
          fullName = p['full_name'] ?? 'مستخدم';
          role = p['role'] ?? 'guest';
        });
      } catch (_) {}
    } else {
      if (mounted) setState(() {
        role = 'guest';
        fullName = 'ضيف';
      });
    }
    try {
      final ann = await withRetry(() => supabase.from('announcements').select('id, title, body, image_url, created_at').order('created_at', ascending: false).limit(10));
      if (mounted) setState(() {
        announcements = ann;
        galleryImages = ann.where((e) => (e['image_url'] ?? '').toString().isNotEmpty).toList();
      });
    } catch (_) {}
    try {
      final s = await withRetry(() => supabase.from('students').select('id'));
      final t = await withRetry(() => supabase.from('teachers').select('id'));
      if (mounted) setState(() {
        studentsCount = s.length;
        teachersCount = t.length;
      });
    } catch (_) {}
  }

class GenericListPage extends StatefulWidget {

  final String subField;
  final IconData icon;

  const GenericListPage({super.key, required this.title, required this.table, required this.titleField, required this.subField, required this.icon, this.isGuest = false});
  @override
  State<GenericListPage> createState() => _GenericListPageState();
}

class _GenericListPageState extends State<GenericListPage> {
  List items = [];
  String msg = 'جارٍ التحميل...';
  @override
  void initState() {
    super.initState();
    load();
  }

  Future<void> load() async {
    try {
      final data = await withRetry(() => supabase.from(widget.table).select('*').order('created_at', ascending: false).limit(50));
      if (mounted) setState(() {
        items = data;
        msg = items.isEmpty ? 'لا توجد بيانات بعد في ${widget.title}.' : '';
      });
    } catch (e) {
      if (mounted) setState(() => msg = 'لا توجد بيانات بعد أو الجدول غير موجود.');
    }
  }

  Future<void> deleteItem(dynamic it) async {
    if (widget.isGuest) {
      ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('الضيف لا يمكنه الحذف')));
      return;
    }
    final ok = await showDialog<bool>(context: context, builder: (_) => AlertDialog(title: const Text('حذف؟'), content: const Text('هل تريد حذف هذا المنشور نهائياً؟'), actions: [TextButton(onPressed: () => Navigator.pop(context, false), child: const Text('لا')), TextButton(onPressed: () => Navigator.pop(context, true), child: const Text('نعم، احذف', style: TextStyle(color: Colors.red)))]));
    if (ok == true) {
      try {
        await supabase.from(widget.table).delete().eq('id', it['id']);
        load();
      } catch (e) {
        if (mounted) ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('فشل الحذف: $e')));
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: bgColor,
      appBar: AppBar(title: Text(widget.title), backgroundColor: bgColor, actions: [IconButton(icon: const Icon(Icons.refresh), onPressed: load)]),
      floatingActionButton: widget.isGuest ? null : FloatingActionButton(backgroundColor: mainGreen, onPressed: () async { final ok = await Navigator.push(context, MaterialPageRoute(builder: (_) => AddItemPage(title: widget.title, table: widget.table))); if (ok == true) load(); }, child: const Icon(Icons.add, color: Colors.white)),
      body: items.isEmpty
          ? Center(child: Padding(padding: const EdgeInsets.all(20), child: Column(mainAxisAlignment: MainAxisAlignment.center, children: [Icon(widget.icon, size: 60, color: mainGreen), const SizedBox(height: 12), Text(msg, textAlign: TextAlign.center), const SizedBox(height: 12), ElevatedButton(onPressed: load, child: const Text('إعادة المحاولة'))])))
          : ListView.builder(
              padding: const EdgeInsets.all(12),
              itemCount: items.length,
              itemBuilder: (_, i) {
                final it = items[i];
                final titleVal = (it[widget.titleField] ?? it['title'] ?? it['full_name'] ?? it['name'] ?? 'بدون عنوان').toString();
                final subVal = (it[widget.subField] ?? it['body'] ?? it['description'] ?? it['status'] ?? '').toString();
                final img = (it['image_url'] ?? '').toString();
                return Container(
                  margin: const EdgeInsets.only(bottom: 10),
                  padding: const EdgeInsets.all(14),
                  decoration: BoxDecoration(color: Colors.white, borderRadius: BorderRadius.circular(14)),
                  child: Column(
                    crossAxisAlignment: CrossAxisAlignment.start,
                    children: [
                      Row(
                        children: [
                          CircleAvatar(backgroundColor: const Color(0xFFE8F0E8), child: Icon(widget.icon, color: mainGreen, size: 20)),
                          const SizedBox(width: 12),
                          Expanded(child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [Text(titleVal, style: const TextStyle(fontWeight: FontWeight.bold, fontSize: 16)), if (subVal.isNotEmpty) Text(subVal, style: const TextStyle(color: Colors.grey, fontSize: 12), maxLines: 2, overflow: TextOverflow.ellipsis)])),
                          if (!widget.isGuest) IconButton(icon: const Icon(Icons.delete, color: Colors.red, size: 22), onPressed: () => deleteItem(it)),
                        ],
                      ),
                      if (img.isNotEmpty && img.startsWith('http'))

class AddItemPage extends StatefulWidget {
  final String title;
  final String table;
  const AddItemPage({super.key, required this.title, required this.table});
  @override
  State<AddItemPage> createState() => _AddItemPageState();
}

class _AddItemPageState extends State<AddItemPage> {
  final f1 = TextEditingController();
  final f2 = TextEditingController();
  final f3 = TextEditingController();
  String msg = '';
  bool loading = false;
  File? pickedImage;

  String get label1 {
    if (widget.table == 'grades') return 'اسم الطالب';
    if (widget.table == 'attendance') return 'اسم الطالب';
    return 'العنوان';
  }

  String get label2 {
    if (widget.table == 'grades') return 'المادة';
    if (widget.table == 'attendance') return 'الحالة (حاضر / غائب)';
    if (widget.table == 'announcements') return 'نص الإعلان';
    return 'الوصف / التفاصيل';
  }

  String get label3 {
    if (widget.table == 'grades') return 'الدرجة';
    if (widget.table == 'attendance') return 'التاريخ';
    return '';
  }

  bool get showImageButton => widget.table == 'homework' || widget.table == 'announcements' || widget.table == 'exams' || widget.table == 'schedule';

  Future<void> pickImage() async {
    try {
      final XFile? x = await ImagePicker().pickImage(source: ImageSource.gallery, imageQuality: 70);
      if (x != null) setState(() => pickedImage = File(x.path));
    } catch (e) {
      if (mounted) setState(() => msg = 'فشل اختيار الصورة: $e');
    }
  }

  Future<String?> uploadImage() async {
    if (pickedImage == null) return null;
    try {
      final name = 'img_${DateTime.now().millisecondsSinceEpoch}.jpg';
      await supabase.storage.from('homework-images').upload(name, pickedImage!);
      return supabase.storage.from('homework-images').getPublicUrl(name);
    } catch (e) {
      if (mounted) setState(() => msg = 'فشل رفع الصورة: $e');
      return null;
    }
  }

  Future<void> publish() async {
    setState(() {
      loading = true;
      msg = '';
    });
    try {
      Map<String, dynamic> data = {'school_id': 'fbf0b515-60f6-4b77-b6e3-238ece1d0883'};
      if (widget.table == 'announcements') {
        data['title'] = f1.text.trim();
        data['body'] = f2.text.trim();
      } else if (widget.table == 'homework') {
        data['title'] = f1.text.trim();
        data['description'] = f2.text.trim();
      } else if (widget.table == 'grades') {
        data['title'] = '${f1.text.trim()} - ${f2.text.trim()}';
        data['description'] = 'الدرجة: ${f3.text.trim()}';
      } else if (widget.table == 'attendance') {
        data['title'] = f1.text.trim();
        data['description'] = 'الحالة: ${f2.text.trim()} - التاريخ: ${f3.text.trim()}';
      } else {
        data['title'] = f1.text.trim();
        data['description'] = f2.text.trim();
      }
      final imgUrl = await uploadImage();
      if (imgUrl != null) data['image_url'] = imgUrl;
      await supabase.from(widget.table).insert(data);
      if (!mounted) return;
      Navigator.pop(context, true);
    } catch (e) {
      if (mounted) setState(() => msg = 'فشل النشر: $e');
    }
    if (mounted) setState(() => loading = false);
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: bgColor,
      appBar: AppBar(title: Text('نشر في ${widget.title}'), backgroundColor: bgColor),
      body: SingleChildScrollView(
        padding: const EdgeInsets.all(20),
        child: Column(
          children: [
            TextField(controller: f1, decoration: InputDecoration(labelText: label1, filled: true, fillColor: Colors.white, border: const OutlineInputBorder(borderRadius: BorderRadius.all(Radius.circular(12))))),
            const SizedBox(height: 12),
            TextField(controller: f2, maxLines: widget.table == 'announcements' ? 4 : 1, decoration: InputDecoration(labelText: label2, filled: true, fillColor: Colors.white, border: const OutlineInputBorder(borderRadius: BorderRadius.all(Radius.circular(12))))),
            if (label3.isNotEmpty) ...[
              const SizedBox(height: 12),
              TextField(controller: f3, decoration: InputDecoration(labelText: label3, filled: true, fillColor: Colors.white, border: const OutlineInputBorder(borderRadius: BorderRadius.all(Radius.circular(12))))),
            ],
            if (showImageButton) ...[
              const SizedBox(height: 16),
              if (pickedImage != null) ClipRRect(borderRadius: BorderRadius.circular(12), child: Image.file(pickedImage!, height: 200, width: double.infinity, fit: BoxFit.cover)),
              const SizedBox(height: 8),
              SizedBox(
                width: double.infinity,
                height: 50,
                child: OutlinedButton.icon(style: OutlinedButton.styleFrom(side: const BorderSide(color: mainGreen), shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(30))), onPressed: pickImage, icon: const Icon(Icons.photo, color: mainGreen), label: Text(pickedImage == null ? 'اختر صورة من الهاتف - للنشر في المعرض' : 'تغيير الصورة', style: const TextStyle(color: mainGreen, fontWeight: FontWeight.bold))),
              ),
            ],
            const SizedBox(height: 20),
            SizedBox(width: double.infinity, height: 50, child: ElevatedButton(style: ElevatedButton.styleFrom(backgroundColor: mainGreen, shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(30))), onPressed: loading ? null : publish, child: Text(loading ? 'جارٍ الرفع والنشر...' : 'نشر الآن', style: const TextStyle(fontSize: 16, fontWeight: FontWeight.bold)))),
            if (msg.isNotEmpty) Padding(padding: const EdgeInsets.only(top: 12), child: Text(msg, textAlign: TextAlign.center, style: const TextStyle(color: Colors.red))),
          ],
        ),
      ),
    );
  }
}

                  padding: const EdgeInsets.all(14),
                  decoration: BoxDecoration(color: Colors.white, borderRadius: BorderRadius.circular(14)),
                  child: Row(
                    children: [
                      const CircleAvatar(backgroundColor: Color(0xFFE8F0E8), child: Icon(Icons.class_, color: mainGreen)),
                      const SizedBox(width: 12),
                      Expanded(child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [Text(s['name'] ?? '', style: const TextStyle(fontWeight: FontWeight.bold)), Text('السعة: ${s['capacity'] ?? '-'}', style: const TextStyle(color: Colors.grey, fontSize: 12))])),
                      if (!widget.isGuest)
                        IconButton(icon: const Icon(Icons.delete, color: Colors.red), onPressed: () async {
                          await supabase.from('sections').delete().eq('id', s['id']);
                          loadSections();
                        }),
                    ],
                  ),
                );
              },
            ),
    );
